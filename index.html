// 언어 자동 감지 헬퍼 function detectLanguage(text) { if (/[\u3040-\u30ff\u31f0-\u31ff]/.test(text)) return 'ja'; // 일본어 가나 if (/[\u4e00-\u9fa5]/.test(text)) return 'zh'; // 한자(중국어) if (/[\uac00-\ud7af\u1100-\u11ff]/.test(text)) return 'ko'; // 한글 if (/[\u0600-\u06ff]/.test(text)) return 'ar'; // 아랍어 if (/[\u0e00-\u0e7f]/.test(text)) return 'th'; // 태국어 if (/[a-zA-Z]/.test(text)) return 'en'; // 영문/라틴 기본 return 'en'; } function speakText(text, langCode) { try { if (!("speechSynthesis" in window)) return; window.speechSynthesis.cancel(); // 지정된 언어가 맞지 않거나 직접 텍스트를 판별해야 할 경우 자동 감지 const detected = detectLanguage(text); const finalLang = SPEECH_LOCALES[langCode] ? langCode : detected; const utter = new SpeechSynthesisUtterance(text); utter.lang = SPEECH_LOCALES[finalLang] || SPEECH_LOCALES[detected] || "en-US"; window.speechSynthesis.speak(utter); } catch (e) {} }
